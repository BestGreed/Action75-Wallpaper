cmake_minimum_required(VERSION 3.16)
project(Action75KeyGame LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(ACTION75_SDK_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../../skills/action75-wallpaper-author/sdk" CACHE PATH "Directory containing Action75Wallpaper.h")
set(ACTION75_SDK_VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/../../../vendor-source/windows-native" CACHE PATH "Directory containing nlohmann/json.hpp")
if(NOT EXISTS "${ACTION75_SDK_INCLUDE}/Action75Wallpaper.h")
  message(FATAL_ERROR "Set ACTION75_SDK_INCLUDE to the directory containing Action75Wallpaper.h.")
endif()
if(NOT EXISTS "${ACTION75_SDK_VENDOR}/nlohmann/json.hpp")
  message(FATAL_ERROR "nlohmann/json.hpp is required. Use the complete source archive or set ACTION75_SDK_VENDOR to the include directory for nlohmann/json 3.12.0.")
endif()
add_library(wallpaper SHARED source/wallpaper.cpp source/wallpaper.def)
target_compile_definitions(wallpaper PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(wallpaper PRIVATE "${ACTION75_SDK_INCLUDE}" "${ACTION75_SDK_VENDOR}")
set_target_properties(wallpaper PROPERTIES PREFIX "")
if(MINGW)
  target_link_options(wallpaper PRIVATE -static -static-libstdc++ -static-libgcc)
endif()
install(TARGETS wallpaper RUNTIME DESTINATION .)
install(FILES pack/manifest.json pack/settings.json DESTINATION .)
include(CTest)
if(BUILD_TESTING)
  add_executable(abi_test source/abi_test.cpp)
  target_compile_definitions(abi_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_include_directories(abi_test PRIVATE "${ACTION75_SDK_INCLUDE}")
  target_link_libraries(abi_test PRIVATE d3d11)
  if(MINGW)
    target_link_options(abi_test PRIVATE -municode)
  endif()
  add_test(NAME game-abi-warp COMMAND abi_test $<TARGET_FILE:wallpaper>)
  add_executable(simulation_test source/simulation_test.cpp)
  add_test(NAME game-simulation COMMAND simulation_test)
endif()
