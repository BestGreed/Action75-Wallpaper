cmake_minimum_required(VERSION 3.20)
project(Action75TidalSignal LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(ACTION75_SDK_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../../skills/action75-wallpaper-author/sdk" CACHE PATH "Public Action75 ABI headers")
set(ACTION75_SDK_VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/../../../vendor-source/windows-native" CACHE PATH "Directory containing nlohmann/json.hpp")
if(NOT EXISTS "${ACTION75_SDK_INCLUDE}/Action75Wallpaper.h" OR NOT EXISTS "${ACTION75_SDK_INCLUDE}/Action75Wallpaper12.h")
  message(FATAL_ERROR "Set ACTION75_SDK_INCLUDE to the directory containing both Action75 ABI headers.")
endif()
if(NOT EXISTS "${ACTION75_SDK_VENDOR}/nlohmann/json.hpp")
  message(FATAL_ERROR "nlohmann/json.hpp is required. Use the complete source archive or set ACTION75_SDK_VENDOR to the include directory for nlohmann/json 3.12.0.")
endif()
add_executable(compile_shader src/compile_shader.cpp)
target_link_libraries(compile_shader PRIVATE d3dcompiler)
foreach(stage vs ps)
  set(shader "${CMAKE_CURRENT_BINARY_DIR}/tide-${stage}.cso")
  add_custom_command(OUTPUT "${shader}" COMMAND compile_shader "${CMAKE_CURRENT_SOURCE_DIR}/src/tide.hlsl" "${stage}" "${shader}"
    DEPENDS compile_shader src/tide.hlsl VERBATIM)
  list(APPEND shaders "${shader}")
endforeach()
add_custom_target(tide_shaders DEPENDS ${shaders})
add_library(wallpaper SHARED src/beacon.cpp)
add_dependencies(wallpaper tide_shaders)
target_compile_definitions(wallpaper PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN ACTION75_WALLPAPER12_EXPORTS)
target_include_directories(wallpaper PRIVATE "${ACTION75_SDK_INCLUDE}" "${ACTION75_SDK_VENDOR}")
target_link_libraries(wallpaper PRIVATE d3d12)
install(TARGETS wallpaper RUNTIME DESTINATION .)
install(FILES ${shaders} DESTINATION shaders)
install(FILES pack/manifest.json pack/settings.json DESTINATION .)
include(CTest)
if(BUILD_TESTING)
  add_executable(verify_beacon src/verify.cpp)
  target_include_directories(verify_beacon PRIVATE "${ACTION75_SDK_INCLUDE}")
  target_compile_definitions(verify_beacon PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(verify_beacon PRIVATE d3d12 dxgi)
  add_test(NAME assemble_beacon COMMAND ${CMAKE_COMMAND} --install ${CMAKE_BINARY_DIR} --config $<CONFIG> --prefix ${CMAKE_BINARY_DIR}/test-pack)
  set_tests_properties(assemble_beacon PROPERTIES FIXTURES_SETUP package)
  add_test(NAME verify_beacon_rgba COMMAND verify_beacon ${CMAKE_BINARY_DIR}/test-pack)
  set_tests_properties(verify_beacon_rgba PROPERTIES FIXTURES_REQUIRED package)
endif()
