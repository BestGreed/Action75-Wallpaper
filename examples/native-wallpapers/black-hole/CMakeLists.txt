cmake_minimum_required(VERSION 3.18)
project(EventHorizonWallpaper LANGUAGES CXX)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(ACTION75_SDK "${REPO}/skills/action75-wallpaper-author/sdk" CACHE PATH "Public Action75 wallpaper SDK headers")
set(JSON_INCLUDE "${REPO}/vendor-source/windows-native/nlohmann" CACHE PATH "Directory containing nlohmann json.hpp")
include(cmake/BlackHoleShaders.cmake)
add_library(wallpaper SHARED src/Plugin.cpp src/BlackHoleDx12.cpp)
action75_black_hole_shaders(wallpaper DX12)
target_include_directories(wallpaper PRIVATE "${ACTION75_SDK}" "${JSON_INCLUDE}")
target_compile_features(wallpaper PRIVATE cxx_std_17)
target_compile_definitions(wallpaper PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX ACTION75_WALLPAPER12_EXPORTS)
target_compile_options(wallpaper PRIVATE /utf-8 /W4 /EHsc)
target_link_libraries(wallpaper PRIVATE d3d12 windowscodecs ole32)
set_property(TARGET wallpaper PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_options(wallpaper PRIVATE /Brepro)
install(TARGETS wallpaper RUNTIME DESTINATION .)
install(FILES pack/manifest.json pack/settings.json pack/nebula.png pack/preview.png
              pack/nebula-source.txt DESTINATION .)

add_executable(verify_native_parity EXCLUDE_FROM_ALL tests/VerifyNativeParity.cpp)
target_compile_features(verify_native_parity PRIVATE cxx_std_17)
target_compile_definitions(verify_native_parity PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(verify_native_parity PRIVATE "${ACTION75_SDK}" "${JSON_INCLUDE}")
target_link_libraries(verify_native_parity PRIVATE d3d11 d3d12 dxgi ole32)
