cmake_minimum_required(VERSION 3.18)
project(EventHorizonWallpaper LANGUAGES CXX)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(ACTION75_SDK "${REPO}/native/windows-player/include" CACHE PATH "Directory containing Action75Wallpaper.h")
set(JSON_INCLUDE "${REPO}/vendor-source/windows-native/nlohmann" CACHE PATH "Directory containing nlohmann json.hpp")
include(cmake/BlackHoleShaders.cmake)
add_library(wallpaper SHARED src/Plugin.cpp src/BlackHoleScene.cpp)
action75_black_hole_shaders(wallpaper)
target_include_directories(wallpaper PRIVATE "${ACTION75_SDK}" "${JSON_INCLUDE}")
target_compile_features(wallpaper PRIVATE cxx_std_17)
target_compile_definitions(wallpaper PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(wallpaper PRIVATE /utf-8 /W4 /EHsc)
target_link_libraries(wallpaper PRIVATE d3d11 windowscodecs ole32)
set_property(TARGET wallpaper PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_custom_command(TARGET wallpaper POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
                   "$<TARGET_FILE:wallpaper>" "${CMAKE_CURRENT_SOURCE_DIR}/pack/wallpaper.dll")
target_link_options(wallpaper PRIVATE /EXPORT:Action75GetWallpaper /Brepro)
add_executable(verify_black_hole EXCLUDE_FROM_ALL src/Verify.cpp)
target_compile_features(verify_black_hole PRIVATE cxx_std_17)
target_include_directories(verify_black_hole PRIVATE "${ACTION75_SDK}")
target_link_libraries(verify_black_hole PRIVATE d3d11 ole32)
add_dependencies(verify_black_hole wallpaper)
