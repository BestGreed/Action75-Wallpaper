cmake_minimum_required(VERSION 3.16)
project(Action75Ferrofluid LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

set(ACTION75_SDK_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../../skills/action75-wallpaper-author/sdk" CACHE PATH "Public Action75 ABI headers")
set(ACTION75_SDK_VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/../../../vendor-source/windows-native" CACHE PATH "Directory containing nlohmann/json.hpp")
if(NOT EXISTS "${ACTION75_SDK_INCLUDE}/Action75Wallpaper12.h" OR NOT EXISTS "${ACTION75_SDK_INCLUDE}/Action75Wallpaper.h")
    message(FATAL_ERROR "Set ACTION75_SDK_INCLUDE to the directory containing the public Action75 ABI headers.")
endif()
if(NOT EXISTS "${ACTION75_SDK_VENDOR}/nlohmann/json.hpp")
    message(FATAL_ERROR "Set ACTION75_SDK_VENDOR to the include directory for nlohmann/json 3.12.0.")
endif()

add_library(wallpaper SHARED Source/Studio.cpp)
set_target_properties(wallpaper PROPERTIES PREFIX "")
target_compile_definitions(wallpaper PRIVATE ACTION75_WALLPAPER12_EXPORTS NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(wallpaper PRIVATE "${ACTION75_SDK_INCLUDE}" "${ACTION75_SDK_VENDOR}")
target_link_libraries(wallpaper PRIVATE d3d12)

add_executable(compile_shader Source/compile_shader.cpp)
target_link_libraries(compile_shader PRIVATE d3dcompiler)
if(MINGW)
    target_link_options(wallpaper PRIVATE -static -static-libstdc++ -static-libgcc)
    target_link_options(compile_shader PRIVATE -municode)
endif()
set(shader_outputs)
foreach(entry IN ITEMS floorVS floorPS fluidVS fluidPS)
    if(entry MATCHES "VS$")
        set(profile vs_5_0)
    else()
        set(profile ps_5_0)
    endif()
    set(output "${CMAKE_CURRENT_BINARY_DIR}/shaders/${entry}.cso")
    add_custom_command(OUTPUT "${output}"
        COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/shaders"
        COMMAND compile_shader "${CMAKE_CURRENT_SOURCE_DIR}/Source/Bath.hlsl" "${entry}" "${profile}" "${output}"
        DEPENDS compile_shader Source/Bath.hlsl
        VERBATIM)
    list(APPEND shader_outputs "${output}")
endforeach()
add_custom_target(ferrofluid_shaders DEPENDS ${shader_outputs})
add_dependencies(wallpaper ferrofluid_shaders)

install(TARGETS wallpaper RUNTIME DESTINATION .)
install(FILES Ferrofluid/manifest.json Ferrofluid/settings.json Ferrofluid/preview.png ${shader_outputs} DESTINATION .)

include(CTest)
if(BUILD_TESTING)
    add_executable(solver_audit Source/solver_audit.cpp)
    add_test(NAME ferrofluid-solver-mass COMMAND solver_audit)
    add_executable(visual_check Source/visual_check.cpp)
    target_compile_definitions(visual_check PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_include_directories(visual_check PRIVATE "${ACTION75_SDK_INCLUDE}" "${ACTION75_SDK_VENDOR}")
    target_link_libraries(visual_check PRIVATE d3d12 dxgi)
    if(MINGW)
        target_link_options(visual_check PRIVATE -municode)
    endif()
endif()
