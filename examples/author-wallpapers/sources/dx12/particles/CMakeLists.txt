cmake_minimum_required(VERSION 3.24)
project(Action75AuthorParticles12 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(AUTHOR_PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(ACTION75_SDK_INCLUDE "${AUTHOR_PROJECT_ROOT}/skills/action75-wallpaper-author/sdk" CACHE PATH "Public ABI headers")
set(NLOHMANN_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party" CACHE PATH "Public nlohmann/json include directory")
set(EFFEKSEER_ROOT "" CACHE PATH "Extracted public EffekseerForCpp1.80.8 source SDK")
set(AUTHOR_PACKS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/assets" CACHE PATH "Preserved original author pack assets")
option(A75_PARTICLE_TESTS "Build independent offscreen DX11/DX12 comparison executable" OFF)
if(NOT EXISTS "${ACTION75_SDK_INCLUDE}/Action75Wallpaper12.h" OR NOT EXISTS "${NLOHMANN_INCLUDE_DIR}/nlohmann/json.hpp")
  message(FATAL_ERROR "Set ACTION75_SDK_INCLUDE to the public ABI2 SDK (nlohmann/json is bundled in third_party)")
endif()
if(NOT EXISTS "${EFFEKSEER_ROOT}/src/EffekseerRendererDX12/EffekseerRendererDX12.h")
  message(FATAL_ERROR "Set EFFEKSEER_ROOT to the extracted EffekseerForCpp1.80.8 source SDK")
endif()
foreach(option BUILD_EXAMPLES BUILD_EDITOR BUILD_VIEWER BUILD_TEST BUILD_EXAMPLE BUILD_TOOL BUILD_GL BUILD_DX9 BUILD_VULKAN BUILD_METAL BUILD_WEBGPU USE_OPENAL USE_XAUDIO2 USE_DSOUND USE_OSM USE_LIBPNG_LOADER)
  set(${option} OFF CACHE BOOL "Not part of the author particle runtime" FORCE)
endforeach()
set(BUILD_DX12 ON CACHE BOOL "Real D3D12 particle renderer" FORCE)
set(BUILD_DX11 ${A75_PARTICLE_TESTS} CACHE BOOL "Comparison test only, never linked to the package DLL" FORCE)
set(USE_MSVC_RUNTIME_LIBRARY_DLL OFF CACHE BOOL "Static MSVC runtime" FORCE)
add_subdirectory("${EFFEKSEER_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/effekseer" EXCLUDE_FROM_ALL)
add_library(author_particles12 SHARED ParticleWallpaper.cpp)
set_target_properties(author_particles12 PROPERTIES OUTPUT_NAME wallpaper)
target_compile_definitions(author_particles12 PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN ACTION75_WALLPAPER12_EXPORTS)
target_include_directories(author_particles12 PRIVATE "${ACTION75_SDK_INCLUDE}" "${NLOHMANN_INCLUDE_DIR}")
target_link_libraries(author_particles12 PRIVATE EffekseerRendererDX12 d3d12 dxgi dxguid d3dcompiler)
foreach(id lantern-almanac tidemark-bearing)
  install(TARGETS author_particles12 RUNTIME DESTINATION "${id}")
  install(DIRECTORY "${AUTHOR_PACKS_ROOT}/${id}/" DESTINATION "${id}"
    PATTERN wallpaper.dll EXCLUDE PATTERN manifest.json EXCLUDE PATTERN particles.json EXCLUDE)
  install(FILES "metadata/${id}/manifest.json" "metadata/${id}/particles.json" DESTINATION "${id}")
  install(DIRECTORY "licenses/" DESTINATION "${id}/licenses")
endforeach()
if(A75_PARTICLE_TESTS)
  enable_testing()
  add_executable(verify_particles Verify.cpp)
  target_compile_definitions(verify_particles PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_include_directories(verify_particles PRIVATE "${ACTION75_SDK_INCLUDE}" "${NLOHMANN_INCLUDE_DIR}")
  target_link_libraries(verify_particles PRIVATE EffekseerRendererDX11 EffekseerRendererDX12 d3d11 d3d12 dxgi dxguid d3dcompiler windowscodecs ole32)
  add_executable(verify_particle_opacity VerifyOpacity.cpp)
  target_compile_definitions(verify_particle_opacity PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_include_directories(verify_particle_opacity PRIVATE "${ACTION75_SDK_INCLUDE}" "${NLOHMANN_INCLUDE_DIR}")
  target_link_libraries(verify_particle_opacity PRIVATE EffekseerRendererDX11 EffekseerRendererDX12 d3d11 d3d12 dxgi dxguid d3dcompiler windowscodecs ole32)
endif()
