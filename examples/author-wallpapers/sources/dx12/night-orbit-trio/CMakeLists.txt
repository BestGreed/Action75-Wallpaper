cmake_minimum_required(VERSION 3.24)
project(NightOrbitTrioDX12 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Night Orbit Trio targets Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options($<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
set(ACTION75_SDK_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/sdk" CACHE PATH "Public ABI1/ABI2 SDK directory")
set(ACTION75_SDK_VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/vendor" CACHE PATH "nlohmann/json.hpp directory")
set(ACTION75_PACK "${CMAKE_CURRENT_SOURCE_DIR}/pack" CACHE PATH "Package metadata directory")
add_executable(compile_shaders src/CompileShaders.cpp)
target_link_libraries(compile_shaders PRIVATE d3dcompiler)
set(shader_dir "${CMAKE_CURRENT_BINARY_DIR}/shaders")
set(shader_files "${shader_dir}/spark-vs.cso" "${shader_dir}/screen-vs.cso" "${shader_dir}/spark-ps.cso"
  "${shader_dir}/blur-ps.cso" "${shader_dir}/composite-ps.cso" "${shader_dir}/idle-ps.cso")
add_custom_command(OUTPUT ${shader_files} "${shader_dir}/FireworksShaders.h"
  COMMAND compile_shaders "${CMAKE_CURRENT_SOURCE_DIR}/src/Fireworks.hlsl" "${shader_dir}"
  DEPENDS compile_shaders src/Fireworks.hlsl src/LayerColor.hlsli VERBATIM)
add_custom_target(shaders DEPENDS ${shader_files} "${shader_dir}/FireworksShaders.h")
add_library(wallpaper SHARED src/Wallpaper.cpp src/Fireworks12.cpp)
add_dependencies(wallpaper shaders)
target_include_directories(wallpaper PRIVATE src "${ACTION75_SDK_INCLUDE}" "${ACTION75_SDK_VENDOR}")
target_compile_definitions(wallpaper PRIVATE ACTION75_WALLPAPER12_EXPORTS)
target_link_libraries(wallpaper PRIVATE d3d12)
install(TARGETS wallpaper RUNTIME DESTINATION .)
install(FILES ${shader_files} DESTINATION shaders)
install(FILES "${ACTION75_PACK}/manifest.json" "${ACTION75_PACK}/settings.json" DESTINATION .)
install(FILES LICENSE NOTICE.txt DESTINATION .)
install(FILES vendor/nlohmann-LICENSE.txt DESTINATION . RENAME nlohmann-json-LICENSE.txt)
include(CTest)
if(BUILD_TESTING)
  add_executable(verify_night tests/Verify.cpp tests/reference/FireworksScene.cpp)
  add_dependencies(verify_night shaders)
  target_include_directories(verify_night PRIVATE tests/reference src "${ACTION75_SDK_INCLUDE}" "${ACTION75_SDK_VENDOR}" "${shader_dir}")
  target_link_libraries(verify_night PRIVATE d3d12 d3d11 dxgi windowscodecs ole32)
  add_test(NAME assemble_night COMMAND ${CMAKE_COMMAND} --install "${CMAKE_BINARY_DIR}" --config $<CONFIG> --prefix "${CMAKE_BINARY_DIR}/test-pack")
  set_tests_properties(assemble_night PROPERTIES FIXTURES_SETUP night_pack)
  add_test(NAME night_dx12_matches_legacy COMMAND verify_night "${CMAKE_BINARY_DIR}/test-pack" "${CMAKE_BINARY_DIR}/evidence")
  set_tests_properties(night_dx12_matches_legacy PROPERTIES FIXTURES_REQUIRED night_pack TIMEOUT 180)
endif()
