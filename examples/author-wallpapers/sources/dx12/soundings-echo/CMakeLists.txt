cmake_minimum_required(VERSION 3.24)
project(SoundingsEchoDX12 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(ACTION75_SDK_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/sdk" CACHE PATH "Public ABI1/ABI2 headers only")
set(ACTION75_JSON_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/vendor" CACHE PATH "nlohmann/json include directory")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options($<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
add_executable(compile_soundings_shaders compile_shaders.cpp)
target_link_libraries(compile_soundings_shaders PRIVATE d3dcompiler)
set(shader_dir "${CMAKE_CURRENT_BINARY_DIR}/shaders")
set(shaders)
foreach(name backgroundVS backgroundPS echoVS echoPS)
  list(APPEND shaders "${shader_dir}/${name}.cso")
endforeach()
add_custom_command(OUTPUT ${shaders}
  COMMAND compile_soundings_shaders "${CMAKE_CURRENT_SOURCE_DIR}/echo.hlsl" "${shader_dir}"
  DEPENDS compile_soundings_shaders echo.hlsl VERBATIM)
add_custom_target(soundings_shaders DEPENDS ${shaders})
add_library(wallpaper SHARED wallpaper.cpp)
add_dependencies(wallpaper soundings_shaders)
target_compile_definitions(wallpaper PRIVATE ACTION75_WALLPAPER12_EXPORTS)
target_include_directories(wallpaper PRIVATE "${ACTION75_SDK_INCLUDE}" "${ACTION75_JSON_INCLUDE}")
target_link_libraries(wallpaper PRIVATE d3d12 windowscodecs ole32)
install(TARGETS wallpaper RUNTIME DESTINATION .)
install(FILES ${shaders} DESTINATION shaders)
install(DIRECTORY assets/ DESTINATION .)
include(CTest)
if(BUILD_TESTING)
  add_executable(verify_soundings verify.cpp)
  target_include_directories(verify_soundings PRIVATE "${ACTION75_SDK_INCLUDE}" "${ACTION75_JSON_INCLUDE}")
  target_link_libraries(verify_soundings PRIVATE d3d12 dxgi windowscodecs ole32)
  set(SOUNDINGS_LEGACY_REFERENCE "" CACHE PATH "Optional frozen legacy SpriteBurst implementation, verification only")
  if(SOUNDINGS_LEGACY_REFERENCE)
    target_sources(verify_soundings PRIVATE "${SOUNDINGS_LEGACY_REFERENCE}/SpriteBurst.cpp")
    target_include_directories(verify_soundings PRIVATE "${SOUNDINGS_LEGACY_REFERENCE}")
    target_compile_definitions(verify_soundings PRIVATE SOUNDINGS_LEGACY_COMPARE SOUNDINGS_LEGACY_REFERENCE=L"${SOUNDINGS_LEGACY_REFERENCE}")
    target_link_libraries(verify_soundings PRIVATE d3d11 d3dcompiler)
  endif()
  add_test(NAME assemble_soundings COMMAND ${CMAKE_COMMAND} --install ${CMAKE_BINARY_DIR} --config $<CONFIG> --prefix ${CMAKE_BINARY_DIR}/test-pack)
  set_tests_properties(assemble_soundings PROPERTIES FIXTURES_SETUP soundings_pack)
  add_test(NAME soundings_pixels COMMAND verify_soundings "${CMAKE_BINARY_DIR}/test-pack" "${CMAKE_BINARY_DIR}/evidence")
  set_tests_properties(soundings_pixels PROPERTIES FIXTURES_REQUIRED soundings_pack)
endif()
